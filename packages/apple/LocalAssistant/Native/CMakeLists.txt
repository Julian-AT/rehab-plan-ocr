cmake_minimum_required(VERSION 3.28)
project(RehaAssistantNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT LLAMA_SOURCE)
  message(FATAL_ERROR "LLAMA_SOURCE must identify the verified pinned checkout")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(RehaAssistantNative STATIC Assistant.cpp)
target_include_directories(RehaAssistantNative PUBLIC include)
target_link_libraries(RehaAssistantNative PRIVATE llama)
target_compile_options(RehaAssistantNative PRIVATE -Wall -Wextra -Werror)
