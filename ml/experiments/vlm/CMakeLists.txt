cmake_minimum_required(VERSION 3.20)
project(reha_vlm LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Pass -DLLAMA_SOURCE=<hash-pinned llama.cpp checkout>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} runtime)
add_library(reha-vlm STATIC adapter.cpp)
target_link_libraries(reha-vlm PUBLIC llama mtmd)
target_include_directories(reha-vlm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(reha-vlm-read main.cpp)
target_link_libraries(reha-vlm-read PRIVATE reha-vlm)
target_include_directories(reha-vlm-read PRIVATE ${LLAMA_SOURCE}/vendor)
